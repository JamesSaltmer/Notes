---- SETUP (paste once; order: ~/Notes, then search . and ~, then current dir) -
N=~/Notes; [ -f "$N/Useful.md" ] || N=$(for d in "$PWD" ~; do find "$d" -mindepth 1 -maxdepth 5 -name '.?*' -prune -o -name SEARCH.txt -printf '%d %h\n' 2>/dev/null | sort -n | while read -r dep p; do [ -f "$p/Useful.md" ] && echo "$p"; done; done | head -1); N=${N:-$PWD}; [ -f "$N/Useful.md" ] && echo "Notes: $N" || echo "Notes NOT found - cd into them and paste again, or set N=/path"
s(){ for f in "$N"/*.md; do awk -v q="$(printf '%s' "$1" | tr A-Z a-z)" -v F="${f##*/}" '
/^```/{fence=!fence}
!fence && /^#/{ match($0,/^#+/); l=RLENGTH; if(p && l<=lvl) p=0; if(!p && index(tolower($0),q)){p=1; lvl=l; print "\n==> " F ":" NR} }
p' "$f"; done; }
h(){ grep -n '^#' "$N"/*.md | grep -i -- "$1" | sed "s|$N/||"; }
g(){ grep -rniI --color=always -B2 -A${2:-8} -- "$1" "$N"; }
