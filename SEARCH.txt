---- SETUP (paste once; order: ~/Notes, then search . and ~, then current dir) -
N=~/Notes; [ -f "$N/Useful.md" ] || N=$(for d in "$PWD" ~; do find "$d" -mindepth 1 -maxdepth 5 -name '.?*' -prune -o -name SEARCH.txt -printf '%d %h\n' 2>/dev/null | sort -n | while read -r dep p; do [ -f "$p/Useful.md" ] && echo "$p"; done; done | head -1); N=${N:-$PWD}; [ -f "$N/Useful.md" ] && echo "Notes: $N" || echo "Notes NOT found - cd into them and paste again, or set N=/path"
s(){
  [ -n "$1" ] || { echo 'Usage: s "literal term"' >&2; return 2; }
  for f in "$N"/*.md; do
    [ -f "$f" ] || continue
    awk -v q="$1" -v F="${f##*/}" '
      function emit() {
        if (hit && text != "") printf "\n==> %s:%d\n%s", F, start, text
      }
      BEGIN { q=tolower(q); start=1 }
      {
        marker=substr($0,1,3)
        if (!fence && (marker=="```" || marker=="~~~")) fence=marker
        else if (fence && marker==fence) fence=""
        if (!fence && /^#+[[:space:]]/) {
          emit()
          text=""; start=NR
          match($0,/^#+/); level=RLENGTH
          for (depth in headings) if (depth >= level) delete headings[depth]
          headings[level]=(index(tolower($0),q)>0)
          hit=0
          for (depth in headings) if (headings[depth]) hit=1
        }
        text=text $0 "\n"
        if (index(tolower($0),q)) hit=1
      }
      END { emit() }
    ' "$f"
  done
}
h(){ grep -n '^#' "$N"/*.md | grep -iF -- "$1" | sed "s|$N/||"; }
g(){ grep -niIF --color=always -B2 -A"${2:-8}" -- "$1" "$N"/*.md; }
------------------------------------------------------------------------------

s "term"      show sections matching headings OR content (including code)
              matching a heading also includes its subsections          s "odat"
h "term"      list matching headings with file:line                    h snmp
g "term" [n]  show every matching Markdown line, with context (default 8)
              includes headings and code blocks                          g odat 2
Search terms are case-insensitive literal text, not regular expressions.
After updating these definitions, paste the SETUP block again in existing terminals.

add | less -R to page long output:  s "sql injection" | less -R
open at a line:  less +188 $N/Useful.md
